== settings/sr3.cfg
# Settings for one game of the Sega racing launcher.
# Every key is also a command-line flag with --set <id> or --play <id>:
# steering_lock = 540 is --steering-lock 540.

# Sets the picture options at once: the cabinet's picture, sharper for your
# screen, or for the Steam Deck. Options set after it win.
# Values: original, recommended or steam_deck.
#look =

# The size the game draws at, or native for the cabinet's own. Sharper but
# heavier when larger.
# Values: WIDTHxHEIGHT or native.
render = 1920x1080

# The size the picture is shown at on your screen (Linux).
# Values: WIDTHxHEIGHT.
output = 1920x1080

# How the picture is enlarged to the screen size (Linux).
# Values: fsr, nis, linear or nearest.
upscaler = fsr

# How sharp FSR and NVIDIA Image Scaling make the picture; 0 is sharpest
# (Linux).
# Values: 0 to 20.
sharpness = 5

# Fills the screen; off shows the game in a window (Linux).
# Values: on or off.
fullscreen = on

# How far your wheel turns from full left to full right in the game; the
# cabinets turned 270°.
# Values: 180, 270, 360, 540, 900 or full.
steering_lock = 900

# The strength of the wheel's force feedback, in percent of the game's; 0
# turns it off.
# Values: 0 to 200.
force_feedback = 100

# How far, in percent, a pad's stick moves before the car steers.
# Values: 0 to 30.
deadzone = 5
